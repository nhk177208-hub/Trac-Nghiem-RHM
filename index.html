                return new Promise((resolve, reject) => {
                    if (!window.indexedDB) return reject(new Error('IndexedDB không khả dụng'));
                    const req = indexedDB.open(DB_NAME, 1);
                    req.onupgradeneeded = () => {
                        if (!req.result.objectStoreNames.contains(STORE)) req.result.createObjectStore(STORE);
                    };
                    req.onsuccess = () => resolve(req.result);
                    req.onerror = () => reject(req.error);

        function loadManifestViaScript() {
            return new Promise(resolve => {
                try { delete window.QUIZ_IMPORT_DATA; } catch (e) { window.QUIZ_IMPORT_DATA = undefined; }
                document.querySelectorAll('script[data-quiz-manifest]').forEach(el => el.remove());

                const load = (src, onFail) => {
                    const s = document.createElement('script');
                    s.setAttribute('data-quiz-manifest', '1');
                    s.src = src;
                    s.onload = () => resolve(window.QUIZ_IMPORT_DATA || null);
                    s.onerror = () => { s.remove(); onFail(); };
                    document.head.appendChild(s);
                };
                load('import-manifest.js?_=' + Date.now(), () => load('import-manifest.js', () => resolve(null)));
            });
        }

        async function readManifestData() {
            const isFileProtocol = location.protocol === 'file:';

            if (!isFileProtocol) {
                try {
                    const res = await fetch('import-manifest.json?_=' + Date.now(), { cache: 'no-store' });
                    if (res.ok) return { data: await res.json(), via: 'import-manifest.json' };
                    console.warn('[Auto-import] import-manifest.json trả về HTTP ' + res.status);
                } catch (err) {
                    console.warn('[Auto-import] Không fetch được import-manifest.json:', err);
                }
            }

            const scriptData = await loadManifestViaScript();
            if (scriptData) return { data: scriptData, via: 'import-manifest.js' };

            if (isFileProtocol) {
                console.warn('[Auto-import] Đang chạy offline (file://) nhưng không tìm thấy import-manifest.js');
            }
            return null;
        }

        async function autoImportFromFolder(manual = false) {
            const result = await readManifestData();
            if (!result) {
                if (manual) {
                    showToast(location.protocol === 'file:'
                        ? "Chạy offline cần file import-manifest.js đặt cùng thư mục!"
                        : "Không tìm thấy import-manifest.json trong thư mục!");
                }
                return;
            }

            const parsed = result.data;
            if (!parsed) {
                if (manual) showToast("File manifest rỗng!");
                return;
            }

            if (parsed.coverImage) {
                kv.setItem('quizAppCoverImage', parsed.coverImage);
            }

            let qArray = parsed.quizzes ? parsed.quizzes : (Array.isArray(parsed) ? parsed : null);
            if (!Array.isArray(qArray)) qArray = [];
            const hasExamsKey = Array.isArray(parsed.exams) && parsed.exams.length > 0;
            if (qArray.length === 0 && !hasExamsKey) {
                if (parsed.coverImage) { loadCoverImage(); }
                return;
            }

            const examArray = Array.isArray(parsed.exams) ? parsed.exams : [];
            const combinedArray = qArray.concat(examArray.map(item => Object.assign({}, item, { isExam: true })));

            let quizzes = getSavedQuizzes();
            const localIndex = new Map(quizzes.map((q, i) => [q.id, i]));
            const seenIds = new Set();
            const newOnes = [];
            let addedCount = 0;
            let updatedCount = 0;

            combinedArray.forEach(item => {
                if (!(item && item.name && item.questions)) return;
                if (item.id !== undefined) {
                    if (seenIds.has(item.id)) return;
                    seenIds.add(item.id);
                }
                if (!item.folder) item.folder = item.isExam ? "Thi thử" : "Chung";
                item.folder = String(item.folder).trim().replace(/\s*\/\s*/g, '/');

                if (item.id !== undefined && localIndex.has(item.id)) {
                    const idx = localIndex.get(item.id);
                    const old = quizzes[idx];
                    const merged = Object.assign({}, item);
                    if (old && old.bestResult) merged.bestResult = old.bestResult;
                    if (JSON.stringify(merged) !== JSON.stringify(old)) {
                        quizzes[idx] = merged;
                        updatedCount++;
                    }
                } else {
                    if (item.id === undefined) item.id = Date.now() + Math.floor(Math.random() * 10000);
                    newOnes.unshift(item);
                    addedCount++;
                }
            });

            if (addedCount > 0 || updatedCount > 0) {
                quizzes = newOnes.concat(quizzes);
                kv.setItem('quizAppSavedQuizzes', JSON.stringify(quizzes));
                loadCoverImage();
                renderSavedQuizzes();
                const parts = [];
                if (updatedCount > 0) parts.push(`cập nhật ${updatedCount} bộ đề`);
                if (addedCount > 0) parts.push(`thêm ${addedCount} bộ đề mới`);
                showToast(`Đã tự động ${parts.join(' và ')} từ thư mục!`);
                if (manual) openStorageModal();
            } else {
                if (parsed.coverImage) loadCoverImage();
                if (manual) showToast("Các bộ đề trong máy đã là bản mới nhất, không có gì thay đổi.");
            }
        }

        function addSelectedQuizzesToFolder(folderName) {
            const safeClass = folderName.replace(/[^a-zA-Z0-9]/g, '_');
            const checkboxes = document.querySelectorAll(`.folder-add-cb-${safeClass}:checked`);
            if (checkboxes.length === 0) {
                showToast("Vui lòng chọn ít nhất một bộ đề để thêm!");
                return;
            }
            const quizzes = getSavedQuizzes();
            let count = 0;
            checkboxes.forEach(cb => {
                const quizId = parseInt(cb.value);
                const quiz = quizzes.find(q => q.id === quizId);
                if (quiz) {
                    quiz.folder = folderName;
                    count++;
                }
            });
            if (count > 0) {
                kv.setItem('quizAppSavedQuizzes', JSON.stringify(quizzes));
                showToast(`Đã thêm thành công ${count} bộ đề vào thư mục "${folderName}"!`);

                renderSavedQuizzes();
            }
        }

        function renameFolderPrompt(oldName) {
            const newName = prompt(`Nhập tên mới cho thư mục "${oldName}":`, oldName.split('/').pop());
            if (newName && newName.trim() !== "") {
                const cleanName = newName.trim().replace(/\s*\/\s*/g, '_');
                const parts = oldName.split('/');
                parts[parts.length - 1] = cleanName;
                const newPath = parts.join('/');

                const quizzes = getSavedQuizzes();
                quizzes.forEach(q => {
                    if (q.folder === oldName || q.folder.startsWith(oldName + '/')) {
                        q.folder = q.folder.replace(oldName, newPath);
                    }
                });
                kv.setItem('quizAppSavedQuizzes', JSON.stringify(quizzes));

                let custom = getCustomFolders();
                custom = custom.map(f => f === oldName || f.startsWith(oldName + '/') ? f.replace(oldName, newPath) : f);
                if (!custom.includes(newPath)) custom.push(newPath);
                saveCustomFolders(custom);

                if (currentFolderFilter === oldName || currentFolderFilter.startsWith(oldName + '/')) {
                    currentFolderFilter = currentFolderFilter.replace(oldName, newPath);
                }
                openFolderModal();
                renderSavedQuizzes();
            }
        }

        function deleteFolderPrompt(folderName) {
            if (confirm(`Bạn có chắc chắn muốn xóa thư mục "${folderName}"? Các bộ đề bên trong sẽ được chuyển về thư mục "Chung".`)) {

                if (currentFolderFilter === folderName || currentFolderFilter.startsWith(folderName + '/')) {
                    currentFolderFilter = "ALL";
                }
                openFolderModal();
                renderSavedQuizzes();
            }
        }

            text += `Đáp án: ${q.answerStr || ''}\n`;
            if (q.explanation) {
                text += `Giải thích: ${q.explanation}\n`;
            }

            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(text).then(() => {
                    showToast(`Đã sao chép Câu ${qIndex + 1} vào bộ nhớ tạm!`);
                }).catch(() => {
                    fallbackCopyText(text);
                });
            } else {
                fallbackCopyText(text);
            }
        }

        function fallbackCopyText(text) {
            const textarea = document.createElement('textarea');
            textarea.value = text;
            document.body.appendChild(textarea);
            textarea.select();
            try {
                document.execCommand('copy');
                showToast("Đã sao chép nội dung câu hỏi vào bộ nhớ tạm!");
            } catch (err) {
                showToast("Không thể sao chép tự động.");
            }
            document.body.removeChild(textarea);
        }

        function importExamJsonFiles(e) {
            const fileArray = Array.from(e.target.files || []);
            const totalFiles = fileArray.length;
            if (totalFiles === 0) return;
            e.target.value = '';

            let quizzes = getSavedQuizzes();
            let successCount = 0;
            let processed = 0;

            function finishIfDone() {
                processed++;
                if (processed < totalFiles) return;
                if (successCount > 0) {
                    kv.setItem('quizAppSavedQuizzes', JSON.stringify(quizzes));
                    renderExamList();
                    renderSavedQuizzes();
                    showToast(`Đã nhập thành công ${successCount} đề thi từ ${totalFiles} file!`);
                } else {
                    showToast(`Không nhập được đề thi nào từ ${totalFiles} file. Kiểm tra lại định dạng JSON!`);
                }
            }

            fileArray.forEach(file => {
                const reader = new FileReader();
                reader.onload = function(event) {
                    try {
                        const parsed = JSON.parse(event.target.result);
                        const eArray = parsed && parsed.exams ? parsed.exams : (parsed && parsed.quizzes ? parsed.quizzes : (Array.isArray(parsed) ? parsed : null));
                        if (Array.isArray(eArray)) {
                            eArray.forEach(item => {
                                if (item && item.name && item.questions) {
                                    item.id = Date.now() + Math.floor(Math.random() * 10000);
                                    item.isExam = true;
                                    item.folder = topicName(37);
                                    quizzes.unshift(item);
                                    successCount++;
                                }
                            });
                        } else if (parsed && typeof parsed === 'object' && parsed.name && parsed.questions) {
                            parsed.id = Date.now() + Math.floor(Math.random() * 10000);
                            parsed.isExam = true;
                            parsed.folder = topicName(37);
                            quizzes.unshift(parsed);
                            successCount++;
                        }
                    } catch (err) {
                        console.error("Lỗi đọc file JSON đề thi:", err);
                    }
                    finishIfDone();
                };
                reader.onerror = finishIfDone;
                reader.readAsText(file);
            });
        }

        function resizeBase64Image(base64Str, maxCm = 7) {
            return new Promise((resolve) => {
                const img = new Image();
                img.onload = function() {
                    const maxHeightPx = maxCm * 37.8;
                    let w = img.width;
                    let h = img.height;
                    
                    if (h > maxHeightPx) {
                        w = Math.round((maxHeightPx / h) * w);
                        h = Math.round(maxHeightPx);
                    }
                    
                    const canvas = document.createElement('canvas');
                    canvas.width = w;
                    canvas.height = h;
                    const ctx = canvas.getContext('2d');
                    ctx.drawImage(img, 0, 0, w, h);
                    resolve(canvas.toDataURL('image/jpeg', 0.9));
                };
                img.onerror = () => resolve(base64Str);
                img.src = base64Str;
            });
        }

        function highlightSelectedText() {
            const selection = window.getSelection();
            if (!selection.rangeCount || selection.isCollapsed) {
                showToast("Vui lòng bôi đen (tô) đoạn văn bản cần highlight!");
                return;
            }
            const range = selection.getRangeAt(0);
            const qTextEl = document.getElementById('question-text');
            const optionsContainer = document.getElementById('options-container');

            let targetEl = null;
            if (qTextEl.contains(range.commonAncestorContainer)) {
                targetEl = qTextEl;
            } else if (optionsContainer.contains(range.commonAncestorContainer)) {
                targetEl = range.commonAncestorContainer.nodeType === 3 ? range.commonAncestorContainer.parentElement.closest('button') : range.commonAncestorContainer.closest('button');
            }

            if (!targetEl) {
                showToast("Vui lòng chọn văn bản bên trong câu hỏi hoặc các đáp án!");
                return;
            }

            try {
                const span = document.createElement('mark');
                span.className = 'bg-yellow-200 text-yellow-900 rounded px-1 font-inherit';
                const contents = range.extractContents();
                span.appendChild(contents);
                range.insertNode(span);
                selection.removeAllRanges();

                if (targetEl === qTextEl) {
                    questionStates[currentIndex].customQuestionHtml = qTextEl.innerHTML;
                } else {
                    questionStates[currentIndex].customOptionsHtml = optionsContainer.innerHTML;
                }
                showToast("Đã highlight phần văn bản được chọn!");
            } catch (err) {
                showToast("Không thể highlight vùng chọn này. Hãy bôi đen gọn trong một dòng.");
            }
        }

        // Tăng nhận diện câu hỏi: hỗ trợ câu kết thúc bằng cả dấu hai chấm (:) và dấu hỏi (?)
        function parseTextToQuestions(text) {
            const lines = text.split('\n').map(l => l.trim());
            const parsedQuestions = [];
            let currentQ = null;
            let currentField = null;

            for (let i = 0; i < lines.length; i++) {
                let line = lines[i];
                if (!line) {
                    if (currentField === 'explanation' && currentQ) {
                        currentQ.explanation += "\n";
                    }
                    continue;
                }

                const isExplicitQStart = line.match(/^(Câu|Bài)\s*\d+[\:\.\-\s]?/i) || line.match(/^[1-9]\d*[\.\)]\s+[^\s]/);
                const isOptionLine = line.match(/^[A-D][\.\)\-]\s*/i);
                const isAnsLine = (
                    line.match(/đáp\s*án/i) || 
                    line.match(/đáp\s*số/i) || 
                    line.match(/đúng/i) || 
                    line.match(/^đ\b/i) ||
                    line.match(/^d\b/i)
                ) && currentField !== 'explanation';

                const isExpLine = line.match(/^(giải\s*thích|lời\s*giải|hướng\s*dẫn|giải)[\:\-\s]/i) ||
                    line.match(/^giải\s*thích$/i) ||
                    line.match(/^lời\s*giải$/i) ||
                    line.match(/^hướng\s*dẫn\s*giải$/i) ||
                    line.match(/^giải\s*chi\s*tiết$/i);

                // Nhận diện câu hỏi: Bắt đầu bằng Câu/Bài/Số HOẶC câu hỏi không có tiền tố nhưng kết thúc bằng : hoặc ? trước danh sách đáp án
                if (isExplicitQStart || (!isOptionLine && !isAnsLine && !isExpLine && (currentQ === null || currentQ.answerStr || currentQ.options.length > 0))) {
                    if (currentQ && currentQ.answerStr) parsedQuestions.push(currentQ);
                    let content = line.replace(/^(Câu|Bài)\s*\d+[\:\.\-\s]?\s*/i, '').replace(/^[1-9]\d*[\.\)]\s*/, '').trim();
                    currentQ = { question: content, images: [], explanationImages: [], options: [], answerStr: null, explanation: '', isStarred: false, isHidden: false };
                    currentField = 'question';
                } 
                else if (currentQ && isOptionLine) {
                    currentQ.options.push(line);
                    currentField = 'option';
                } 
                else if (currentQ && isAnsLine) {
                    let letterMatch = line.match(/\b([A-D])\b/i);
                    if (letterMatch) {
                        currentQ.answerStr = letterMatch[1].toUpperCase();
                    }
                    currentField = 'answer';
                } 
                else if (currentQ && isExpLine) {
                    let expText = line.replace(/^(giải\s*thích|lời\s*giải|hướng\s*dẫn\s*giải|hướng\s*dẫn|giải\s*chi\s*tiết|giải)[\:\-\s]*\s*/i, '').trim();
                    currentQ.explanation = expText;
                    currentField = 'explanation';
                }
                else if (currentQ && currentField === 'explanation') {
                    if (currentQ.explanation) {
                        currentQ.explanation += " " + line;
                    } else {
                        currentQ.explanation = line;
                    }
                }
                else if (currentQ && currentField === 'question') {
                    currentQ.question += " " + line;
                }
            }
            if (currentQ && currentQ.answerStr) parsedQuestions.push(currentQ);
            return parsedQuestions;
        }

        function attachImageToQuestion(qIndex, inputElement) {
            const file = inputElement.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(event) {
                const base64Img = event.target.result;
                const quizzes = getSavedQuizzes();
                const quiz = quizzes.find(q => q.id === editingQuizId);
                if (quiz) {
                    if (!quiz.questions[qIndex].images) quiz.questions[qIndex].images = [];
                    if (quiz.questions[qIndex].images.length < 5) {
                        quiz.questions[qIndex].images.push(base64Img);
                        kv.setItem('quizAppSavedQuizzes', JSON.stringify(quizzes));
                        renderEditQuestions();
                    } else {
                        showToast("Chỉ được gắn tối đa 5 ảnh cho mỗi câu hỏi!");
                    }
                }
            };
            reader.readAsDataURL(file);
        }

        async function pasteImageToQuestion(qIndex) {
            try {
                const clipboardItems = await navigator.clipboard.read();
                for (const clipboardItem of clipboardItems) {
                    for (const type of clipboardItem.types) {
                        if (type.startsWith('image/')) {
                            const blob = await clipboardItem.getType(type);
                            const reader = new FileReader();
                            reader.onload = function(event) {
                                const base64Img = event.target.result;
                                const quizzes = getSavedQuizzes();
                                const quiz = quizzes.find(q => q.id === editingQuizId);
                                if (quiz) {
                                    if (!quiz.questions[qIndex].images) quiz.questions[qIndex].images = [];
                                    if (quiz.questions[qIndex].images.length < 5) {
                                        quiz.questions[qIndex].images.push(base64Img);
                                        kv.setItem('quizAppSavedQuizzes', JSON.stringify(quizzes));
                                        renderEditQuestions();
                                        showToast("Đã dán ảnh từ clipboard thành công!");
                                    } else {
                                        showToast("Chỉ được gắn tối đa 5 ảnh cho mỗi câu hỏi!");
                                    }
                                }
                            };
                            reader.readAsDataURL(blob);
                            return;
                        }
                    }
                }
                showToast("Không tìm thấy ảnh trong clipboard!");
            } catch (err) {
                showToast("Hãy sao chép ảnh vào clipboard rồi bấm dán lại!");
            }
        }

        const DIFF_LEVELS = ['easy', 'medium', 'hard'];
        const DIFF_LABEL = { easy: 'Dễ', medium: 'Vừa', hard: 'Khó' };
        const DIFF_STYLE = {
            easy:   { off: 'bg-green-50 text-green-700 border-green-300 hover:bg-green-100',  on: 'bg-green-500 text-white border-green-500' },
            medium: { off: 'bg-amber-50 text-amber-700 border-amber-300 hover:bg-amber-100',  on: 'bg-amber-400 text-white border-amber-400' },
            hard:   { off: 'bg-red-50 text-red-700 border-red-300 hover:bg-red-100',          on: 'bg-red-500 text-white border-red-500' }
        };

        function getDifficultyMap() {
            try { return JSON.parse(kv.getItem('quizAppQuestionDifficulty') || '{}') || {}; } catch (e) { return {}; }
        }
        function getQuestionDifficulty(q, map) {
            map = map || getDifficultyMap();
            const d = map[getCleanQuestionText(q.question)] || q.difficulty;
            return DIFF_LEVELS.includes(d) ? d : '';
        }

        function setCurrentDifficulty(level) {
            const q = questions[currentIndex];
            if (!q) return;
            const key = getCleanQuestionText(q.question);
            const newLevel = getQuestionDifficulty(q) === level ? '' : level;

            const map = getDifficultyMap();
            if (newLevel) map[key] = newLevel; else delete map[key];
            kv.setItem('quizAppQuestionDifficulty', JSON.stringify(map));

            const quizzes = getSavedQuizzes();
            let changed = false;
            quizzes.forEach(qz => (qz.questions || []).forEach(x => {
                if (getCleanQuestionText(x.question) !== key) return;
                if (newLevel) { if (x.difficulty !== newLevel) { x.difficulty = newLevel; changed = true; } }
                else if (x.difficulty) { delete x.difficulty; changed = true; }
            }));
            if (changed) kv.setItem('quizAppSavedQuizzes', JSON.stringify(quizzes));

            if (newLevel) q.difficulty = newLevel; else delete q.difficulty;
            updateDifficultyButtonState();
            showToast(newLevel ? `Đã gắn thẻ: ${DIFF_LABEL[newLevel]}` : 'Đã bỏ thẻ độ khó');
        }

        function updateDifficultyButtonState() {
            const q = questions[currentIndex];
            const cur = q ? getQuestionDifficulty(q) : '';
            document.querySelectorAll('#difficulty-group button').forEach(btn => {
                const lv = btn.dataset.level;
                btn.className = 'eq-btn flex-1 rounded-xl border font-bold text-xs sm:text-sm transition-colors flex items-center justify-center ' +
                    (cur === lv ? DIFF_STYLE[lv].on : DIFF_STYLE[lv].off);
            });
        }

        function openDeepModal() {
            const quizzes = getRegularQuizzes();
            if (quizzes.length === 0) {
                showToast("Chưa có bộ đề nào được lưu để ôn tập!");
                return;
            }
            const groups = {};
            // Không hiện thư mục chưa phân chủ đề trong phần ôn tập chuyên sâu (bỏ t === 0)
            quizzes.forEach(q => { 
                const t = getQuizTopic(q); 
                if (t > 0) {
                    (groups[t] = groups[t] || []).push(q);
                }
            });
            const order = Object.keys(groups).map(Number).sort((a, b) => a - b);

            const tree = document.getElementById('deep-tree');
            tree.innerHTML = '';
            order.forEach(t => {
                const list = groups[t];
                const wrap = document.createElement('div');
                wrap.className = 'bg-white border rounded-xl p-2';
                wrap.innerHTML = `
                    <label class="flex items-center gap-2 font-bold text-gray-800 cursor-pointer">
                        <input type="checkbox" class="deep-topic-cb w-4 h-4 accent-teal-600" data-topic="${t}" checked onchange="toggleDeepTopic(${t}, this.checked)">
                        ${topicFullName(t)} <span class="text-xs font-normal text-gray-500">(${list.length} bộ đề)</span>
                    </label>
                    <div class="mt-1.5 ml-6 space-y-1">
                        ${list.map(q => `
                        <label class="flex items-center gap-2 cursor-pointer hover:bg-teal-50 rounded px-1 py-0.5">
                            <input type="checkbox" class="deep-quiz-cb w-3.5 h-3.5 accent-teal-600" value="${q.id}" data-topic="${t}" checked onchange="syncDeepTopics()">
                            <span class="text-gray-700">${escTopic(q.name)}</span>
                            <span class="text-xs text-gray-400">(${q.questions.length} câu)</span>
                        </label>`).join('')}
                    </div>`;
                tree.appendChild(wrap);
            });
            document.getElementById('deep-modal').classList.remove('hidden');
            updateDeepSummary();
        }

        function deepSelectRandom() {
            const cbs = Array.from(document.querySelectorAll('.deep-quiz-cb'));
            if (cbs.length === 0) return;
            const k = 1 + Math.floor(Math.random() * cbs.length);
            const pick = new Set(shuffleArray(cbs.map(c => c.value)).slice(0, k));

        function getDeepPercents() {
            return DIFF_LEVELS.map(l => {

        function computeDeepPlan(pool, numQ, pct) {
            const sum = pct.reduce((a, b) => a + b, 0);
            const raw = pct.map(p => numQ * p / sum);
            const target = raw.map(Math.floor);
            let rem = numQ - target.reduce((a, b) => a + b, 0);
            raw.map((r, i) => [r - Math.floor(r), i]).sort((a, b) => b[0] - a[0]).slice(0, rem).forEach(x => target[x[1]]++);

            const take = { easy: 0, medium: 0, hard: 0, none: 0 };
            DIFF_LEVELS.forEach((l, i) => { take[l] = Math.min(target[i], pool[l].length); });
            let short = numQ - DIFF_LEVELS.reduce((a, l) => a + take[l], 0);
            const shortBase = short;
            [...DIFF_LEVELS, 'none'].forEach(l => {
                if (short <= 0) return;
                const add = Math.min(pool[l].length - take[l], short);
                take[l] += add;
                short -= add;
            });
            return { take, filled: shortBase, unfilled: short };
        }

        function importTopicJsonFiles(e) {
            const topic = currentTopic;
            const files = Array.from(e.target.files || []);
            e.target.value = '';
            if (files.length === 0 || !topic) return;

            const quizzes = getSavedQuizzes();
            let ok = 0, done = 0, seq = 0;
            const finish = () => {
                if (++done < files.length) return;
                if (ok > 0) {
                    kv.setItem('quizAppSavedQuizzes', JSON.stringify(quizzes));
                    renderSavedQuizzes();
                    showToast(`Đã thêm ${ok} bộ đề vào ${topicName(topic)}!`);
                } else {
                    showToast("Không thêm được bộ đề nào. Kiểm tra lại định dạng JSON!");
                }
            };
            files.forEach(file => {
                const reader = new FileReader();
                reader.onload = ev => {
                    try {
                        const parsed = JSON.parse(ev.target.result);
                        const arr = parsed && parsed.quizzes ? parsed.quizzes
                            : (Array.isArray(parsed) ? parsed
                            : (parsed && parsed.name && parsed.questions ? [parsed] : null));
                        if (Array.isArray(arr)) {
                            arr.forEach(item => {
                                if (item && item.name && Array.isArray(item.questions)) {
                                    item.id = Date.now() + (seq++);
                                    item.folder = topicName(topic);
                                    if (topic === 37) item.isExam = true;
                                    if (!item.date) item.date = new Date().toLocaleDateString('vi-VN');
                                    quizzes.unshift(item);
                                    ok++;
                                }
                            });
                        }
                    } catch (err) {
                        console.error("Lỗi đọc file JSON:", file.name, err);
                    }
                    finish();
                };
                reader.onerror = finish;
                reader.readAsText(file);
            });
        }

/* STREAMING_CHUNK:Configuring randomized question arrays and option shuffling... */
        function shuffleArray(array) {
            let arr = [...array];
            for (let i = arr.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [arr[i], arr[j]] = [arr[j], arr[i]];
            }
            return arr;
        }

        function shuffleQuestionOptions(q) {
            if (!q.options || q.options.length === 0) return q;
            const parsedOptions = q.options.map(opt => {
                const match = opt.match(/^[A-D][\.\)]\s*(.*)$/);
                const content = match ? match[1] : opt.replace(/^[A-D][\.\)]\s*/, '');
                const optLetter = opt.charAt(0).toUpperCase();
                const isCorrect = (optLetter === q.answerStr);
                return { content, isCorrect };
            });
            const shuffled = shuffleArray(parsedOptions);
            const letters = ['A', 'B', 'C', 'D'];
            let newAnswerStr = 'A';
            const newOptions = shuffled.map((item, idx) => {
                const letter = letters[idx] || String.fromCharCode(65 + idx);
                if (item.isCorrect) {
                    newAnswerStr = letter;
                }
                return `${letter}. ${item.content}`;
            });
            return {
                ...q,
                options: newOptions,
                answerStr: newAnswerStr
            };
        }
